Executive Summary
What It Does
Intelligent test case optimization leverages AI to identify redundant, overlapping, and low-value regression tests within the test suite. The system analyzes test coverage relationships, execution history, defect correlation, and code churn patterns to recommend pruning strategies that reduce overall test execution time while maintaining or improving defect detection capability.
Business Value
  • Reduce test execution time by 30-40% through intelligent pruning
  • Improve CI/CD velocity without sacrificing coverage
  • Lower infrastructure costs by reducing test resource consumption
  • Faster feedback loops for developers (< 10 min regression runs)
  • Maintain or improve defect detection through ML-driven coverage analysis

Technical Architecture
System Components
Test Metadata Ingestion
Connectors for pytest, Jest, Selenium Hub; webhooks from CI/CD systems. Extracts test names, assertions, execution time, pass/fail history.
Coverage Analysis Engine
Analyzes coverage.py/Istanbul reports; builds test-to-code relationship graphs; identifies overlapping coverage regions.
ML Scoring Module
scikit-learn/XGBoost model trained on historical test data. Scores redundancy (0-1), low-value (0-1), flakiness (0-1). Outputs test risk matrix.
FastAPI Backend
RESTful API for metadata ingestion, model scoring, recommendation retrieval. Caches model predictions.
PostgreSQL
Stores test metadata, coverage data, execution history, model predictions, user feedback on recommendations.
React Frontend
Visualization: test dependency graphs, redundancy heatmaps, pruning scenarios, execution time savings projections.
Data Flow
CI/CD (pytest/Jest output) ↓ Metadata Ingestion API ↓ PostgreSQL (tests, coverage, history) ↓ Feature Engineering Pipeline ↓ ML Model (scoring redundancy/low-value) ↓ Recommendation Cache ↓ React UI + FastAPI endpoints ↓ User accepts/rejects pruning recommendation ↓ Feedback loop → model retraining

# sentinelqa/backend/ingest.py from fastapi import APIRouter, HTTPException from pydantic import BaseModel from sqlalchemy.orm import Session from datetime import datetime router = APIRouter(prefix="/ingest", tags=["ingest"]) class TestMetadata(BaseModel): name: str suite_id: int duration: float @router.post("/tests") async def ingest_tests(tests: list[TestMetadata], db: Session): """Ingest test metadata from CI/CD""" try: for test in tests: db_test = Tests( name=test.name, suite_id=test.suite_id, avg_duration=test.duration, last_executed=datetime.utcnow() ) db.add(db_test) db.commit() return {"status": "ok", "count": len(tests)} except Exception as e: db.rollback() raise HTTPException(status_code=400, detail=str(e)) @router.post("/coverage") async def ingest_coverage(test_id: int, file: str, coverage: float, db: Session): """Ingest coverage relationship data""" coverage_rel = CoverageRelation( test_id=test_id, code_file=file, coverage_percentage=coverage ) db.add(coverage_rel) db.commit() return {"status": "ok"} @router.post("/execution") async def ingest_execution(test_id: int, passed: bool, duration: float, ci_run_id: str, db: Session): """Ingest test execution history""" execution = ExecutionHistory( test_id=test_id, passed=passed, duration=duration, executed_at=datetime.utcnow(), ci_run_id=ci_run_id ) db.add(execution) db.commit() return {"status": "ok"}
Phase 2: ML Scoring (Weeks 4-6)
2.1 Feature Engineering Pipeline
# sentinelqa/ml/feature_engineering.py import pandas as pd from sklearn.preprocessing import StandardScaler def extract_features(test_id, db_session): """Extract features for ML model""" # Coverage redundancy: % tests covering same code test = db_session.query(Tests).filter_by(id=test_id).first() coverage_files = db_session.query(CoverageRelation)\ .filter_by(test_id=test_id).all() # Calculate coverage overlap overlap_count = 0 for coverage in coverage_files: other_tests = db_session.query(CoverageRelation)\ .filter_by(code_file=coverage.code_file)\ .count() overlap_count += other_tests - 1 redundancy = overlap_count / max(len(coverage_files), 1) # Low-value score: based on execution history executions = db_session.query(ExecutionHistory)\ .filter_by(test_id=test_id)\ .order_by(ExecutionHistory.executed_at.desc())\ .limit(100).all() if executions: pass_rate = sum(1 for e in executions if e.passed) / len(executions) avg_duration = sum(e.duration for e in executions) / len(executions) else: pass_rate = 1.0 avg_duration = 0 # Flakiness: variance in pass rate flakiness = 1.0 - pass_rate if pass_rate < 0.95 else 0.0 # Low-value heuristic: high pass rate + long duration + high overlap low_value_score = (pass_rate * 0.4) + (avg_duration / 100 * 0.3) + (redundancy * 0.3) return { 'redundancy': redundancy, 'low_value_score': low_value_score, 'flakiness': flakiness, 'pass_rate': pass_rate, 'avg_duration': avg_duration } def prepare_training_data(db_session): """Prepare dataset for model training""" tests = db_session.query(Tests).all() features_list = [] for test in tests: features = extract_features(test.id, db_session) features['test_id'] = test.id features_list.append(features) return pd.DataFrame(features_list)
2.2 ML Model Training
# sentinelqa/ml/model.py from sklearn.ensemble import XGBClassifier from sklearn.preprocessing import StandardScaler import joblib class SmartPruningModel: def __init__(self): self.model = XGBClassifier(n_estimators=100, max_depth=6) self.scaler = StandardScaler() def train(self, X, y): """Train model on historical data""" X_scaled = self.scaler.fit_transform(X) self.model.fit(X_scaled, y) def predict_prune_probability(self, X): """Predict if test should be pruned""" X_scaled = self.scaler.transform(X) return self.model.predict_proba(X_scaled)[:, 1] def save(self, path): joblib.dump({ 'model': self.model, 'scaler': self.scaler }, path) def load(self, path): data = joblib.load(path) self.model = data['model'] self.scaler = data['scaler'] # Training script def retrain_model(): from sentinelqa.ml.feature_engineering import prepare_training_data from sentinelqa.db import SessionLocal db = SessionLocal() X = prepare_training_data(db) # Labels: manually marked prunable tests or empirical data y = X['low_value_score'] > 0.7 # Simple heuristic model = SmartPruningModel() model.train(X[['redundancy', 'flakiness', 'avg_duration']], y) model.save('/models/smart_pruning_v1.pkl') db.close()

# sentinelqa/backend/scoring.py from fastapi import APIRouter from sqlalchemy.orm import Session from sentinelqa.ml.feature_engineering import extract_features from sentinelqa.ml.model import SmartPruningModel router = APIRouter(prefix="/score", tags=["scoring"]) model = SmartPruningModel() model.load('/models/smart_pruning_v1.pkl') @router.get("/tests") async def score_all_tests(db: Session): """Score all tests for pruning recommendation""" tests = db.query(Tests).all() results = [] for test in tests: features = extract_features(test.id, db) prune_prob = model.predict_prune_probability( [[features['redundancy'], features['flakiness'], features['avg_duration']]] )[0] # Save recommendation recommendation = PruningRecommendation( test_id=test.id, redundancy_score=features['redundancy'], low_value_score=features['low_value_score'], confidence=prune_prob, reason=f"High overlap ({features['redundancy']:.2%}) + Pass rate {features['pass_rate']:.2%}" ) db.add(recommendation) results.append({ 'test_id': test.id, 'test_name': test.name, 'prune_probability': float(prune_prob), 'reason': recommendation.reason }) db.commit() return results @router.post("/apply") async def apply_pruning(test_ids: list[int], db: Session): """Apply pruning decision and monitor""" pruned_suite = [t.name for t in db.query(Tests).filter(Tests.id.notin_(test_ids)).all()] return { 'status': 'ok', 'pruned_count': len(pruned_suite), 'remaining_tests': len(test_ids), 'estimated_time_saved': '35%' }
