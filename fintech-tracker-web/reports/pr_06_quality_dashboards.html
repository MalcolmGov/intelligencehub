// Kafka Consumer for multi-source data
import json
from kafka import KafkaConsumer
from sqlalchemy import create_engine

class QualityDataConsumer:
    def __init__(self):
        self.consumer = KafkaConsumer(
            'quality-events',
            bootstrap_servers=['kafka:9092'],
            group_id='dashboard-ingestion',
            value_deserializer=lambda m: json.loads(m.decode('utf-8'))
        )
        self.db_engine = create_engine(
            'postgresql://user:pass@postgres:5432/quality_db'
        )

    def normalize_event(self, event):
        """Normalize events from different sources"""
        source = event.get('source_type')

        if source == 'jest':
            return {
                'source': 'jest',
                'test_file': event['test_file'],
                'passed': event['passed'],
                'duration_ms': event['duration'],
                'timestamp': event['timestamp'],
                'team': event.get('team', 'unknown')
            }
        elif source == 'pytest':
            return {
                'source': 'pytest',
                'test_file': event['test_path'],
                'passed': event['outcome'] == 'passed',
                'duration_ms': event['duration_ms'],
                'timestamp': event['timestamp'],
                'team': event.get('team', 'unknown')
            }
        elif source == 'jira':
            return {
                'source': 'jira',
                'issue_type': event['issuetype'],
                'priority': event['priority'],
                'created_date': event['created'],
                'team': event['project'],
                'status': event['status']
            }
        elif source == 'github_deploy':
            return {
                'source': 'deployment',
                'service': event['service'],
                'version': event['version'],
                'deployed_at': event['timestamp'],
                'status': event['status'],
                'team': event['team']
            }
        elif source == 'prometheus':
            return {
                'source': 'production_metrics',
                'metric_name': event['metric'],
                'value': event['value'],
                'timestamp': event['timestamp'],
                'labels': event.get('labels', {})
            }
        elif source == 'elastic':
            return {
                'source': 'logs',
                'error_count': event['error_count'],
                'error_types': event['error_types'],
                'timestamp': event['timestamp'],
                'service': event['service']
            }

    def consume_and_store(self):
        for message in self.consumer:
            normalized = self.normalize_event(message.value)
            self.store_event(normalized)

    def store_event(self, normalized_event):
        with self.db_engine.connect() as conn:
            conn.execute(
                f"""INSERT INTO quality_events
                   (source, event_data, timestamp)
                   VALUES (%s, %s, %s)""",
                (normalized_event['source'],
                 json.dumps(normalized_event),
                 normalized_event.get('timestamp', 'now()'))
            )
            conn.commit()

-- Enable Timescale extension for time-series optimization
CREATE EXTENSION IF NOT EXISTS timescaledb;

CREATE TABLE IF NOT EXISTS quality_events (
    time TIMESTAMP NOT NULL,
    source VARCHAR(50) NOT NULL,
    event_data JSONB NOT NULL,
    event_type VARCHAR(100),
    team VARCHAR(100),
    severity VARCHAR(20)
);

-- Convert to hypertable for efficient time-series queries
SELECT create_hypertable('quality_events', 'time', if_not_exists => TRUE);

-- Create continuous aggregates for performance insights
CREATE MATERIALIZED VIEW daily_quality_metrics AS
SELECT
    time_bucket('1 day', time) as day,
    source,
    team,
    COUNT(*) as event_count,
    ROUND(
        (COUNT(*) FILTER (WHERE event_data->>'passed' = 'true')
         * 100.0 / COUNT(*)), 2
    ) as pass_rate,
    ROUND(
        (event_data->>'duration_ms')::float, 2
    ) as avg_duration_ms
FROM quality_events
WHERE event_type = 'test_result'
GROUP BY time_bucket('1 day', time), source, team;

CREATE INDEX idx_quality_events_team_time
    ON quality_events (team, time DESC);

CREATE INDEX idx_quality_events_source_time
    ON quality_events (source, time DESC);

Forecasting Model (Prophet for Quality Metrics)
import pandas as pd
from fbprophet import Prophet
from sklearn.metrics import mean_absolute_percentage_error
import pickle

class QualityForecastModel:
    def __init__(self, metric_name):
        self.metric_name = metric_name
        self.model = Prophet(
            yearly_seasonality=True,
            weekly_seasonality=True,
            daily_seasonality=False,
            interval_width=0.95
        )

    def prepare_data(self, metrics_df):
        """Format metrics for Prophet"""
        # Aggregate daily metrics
        daily_metrics = metrics_df.groupby('date').agg({
            'pass_rate': 'mean',
            'test_count': 'sum',
            'duration_ms': 'mean',
            'error_count': 'sum'
        }).reset_index()

        # Rename for Prophet compatibility
        daily_metrics.rename(
            columns={'date': 'ds', self.metric_name: 'y'},
            inplace=True
        )

        # Add regressors (exogenous variables)
        daily_metrics['deployment_flag'] = (
            daily_metrics['date'].isin(get_deployment_dates())
        ).astype(int)

        return daily_metrics[['ds', 'y', 'deployment_flag']]

    def train(self, historical_metrics_df):
        """Train forecasting model"""
        df = self.prepare_data(historical_metrics_df)

        # Add external regressor
        self.model.add_regressor('deployment_flag')

        # Fit model
        self.model.fit(df)

        # Validate on holdout set
        future = self.model.make_future_dataframe(periods=14)
        forecast = self.model.predict(future)

        # Calculate validation metric
        validation_mape = mean_absolute_percentage_error(
            df[-30:]['y'].values,
            forecast[-30:-16]['yhat'].values
        )

        return {'validation_mape': validation_mape}

    def forecast(self, periods_ahead=14):
        """Generate forecast for next 2 weeks"""
        future = self.model.make_future_dataframe(periods=periods_ahead)
        forecast = self.model.predict(future)

        return {
            'forecast_dates': forecast['ds'].tail(periods_ahead).tolist(),
            'predicted_values': forecast['yhat'].tail(periods_ahead).tolist(),
            'upper_bound': forecast['yhat_upper'].tail(periods_ahead).tolist(),
            'lower_bound': forecast['yhat_lower'].tail(periods_ahead).tolist(),
        }

    def save(self, path):
        with open(f'{path}/model_{self.metric_name}.pkl', 'wb') as f:
            pickle.dump(self.model, f)
Anomaly Detection with Isolation Forest
from sklearn.ensemble import IsolationForest
import numpy as np

class QualityAnomalyDetector:
    def __init__(self, contamination=0.1):
        self.detector = IsolationForest(
            contamination=contamination,
            random_state=42,
            n_estimators=100
        )

    def extract_features(self, daily_metrics):
        """Create feature vectors from metrics"""
        features = np.column_stack([
            daily_metrics['pass_rate'].values,
            daily_metrics['avg_duration_ms'].values,
            daily_metrics['test_count'].values,
            daily_metrics['error_rate'].values,
            daily_metrics['flaky_test_count'].values,
        ])
        return features

    def train(self, historical_daily_metrics):
        """Train anomaly detector"""
        features = self.extract_features(historical_daily_metrics)
        self.detector.fit(features)

    def detect_anomalies(self, current_daily_metrics):
        """Identify anomalous days"""
        features = self.extract_features(current_daily_metrics)
        predictions = self.detector.predict(features)
        scores = self.detector.score_samples(features)

        anomalies = []
        for idx, pred in enumerate(predictions):
            if pred == -1:  # Anomaly
                anomalies.append({
                    'date': current_daily_metrics.iloc[idx]['date'],
                    'anomaly_score': float(scores[idx]),
                    'metrics': {
                        'pass_rate': float(current_daily_metrics.iloc[idx]['pass_rate']),
                        'duration_ms': float(current_daily_metrics.iloc[idx]['avg_duration_ms']),
                        'error_rate': float(current_daily_metrics.iloc[idx]['error_rate'])
                    }
                })

        return anomalies

from fastapi import FastAPI, BackgroundTasks
from datetime import datetime, timedelta
import asyncio

app = FastAPI()

class InsightGenerator:
    def __init__(self):
        self.forecast_model = QualityForecastModel('pass_rate')
        self.anomaly_detector = QualityAnomalyDetector()

    def generate_daily_insights(self):
        """Generate insights for dashboard"""
        # Get current metrics
        current_metrics = self.fetch_today_metrics()

        # Detect anomalies
        anomalies = self.anomaly_detector.detect_anomalies(
            pd.DataFrame([current_metrics])
        )

        # Generate forecast
        forecast = self.forecast_model.forecast(periods_ahead=14)

        # Analyze trends
        trend_analysis = self.analyze_trends()

        # Create recommendations
        recommendations = self.create_recommendations(
            current_metrics, anomalies, trend_analysis
        )

        return {
            'timestamp': datetime.utcnow().isoformat(),
            'current_state': current_metrics,
            'anomalies': anomalies,
            'forecast': forecast,
            'trends': trend_analysis,
            'recommendations': recommendations,
            'health_score': self.calculate_health_score(current_metrics)
        }

    def analyze_trends(self):
        """Analyze quality trends"""
        metrics_30d = self.fetch_metrics(days=30)

        trends = {
            'pass_rate_trend': self.calculate_slope(metrics_30d['pass_rate']),
            'duration_trend': self.calculate_slope(metrics_30d['avg_duration']),
            'error_trend': self.calculate_slope(metrics_30d['error_rate']),
            'flakiness_trend': self.calculate_slope(metrics_30d['flaky_tests'])
        }

        return {
            'direction': {k: 'improving' if v < 0 else 'degrading'
                         for k, v in trends.items()},
            'magnitude': trends
        }

    def create_recommendations(self, current, anomalies, trends):
        """Generate actionable recommendations"""
        recommendations = []

        if current['pass_rate'] < 0.95:
            recommendations.append({
                'priority': 'high',
                'type': 'test_quality',
                'message': 'Pass rate below 95% - investigate failures',
                'suggested_action': 'Review recent test failures and code changes'
            })

        if trends['duration_trend'] > 0.05:
            recommendations.append({
                'priority': 'medium',
                'type': 'performance',
                'message': 'Test execution time increasing',
                'suggested_action': 'Consider test parallelization or optimization'
            })

        if anomalies:
            recommendations.append({
                'priority': 'high',
                'type': 'anomaly',
                'message': f'{len(anomalies)} anomalies detected',
                'suggested_action': 'Review recent deployments and code changes'
            })

        return recommendations

    def calculate_health_score(self, metrics):
        """Calculate overall quality health (0-100)"""
        components = {
            'pass_rate': metrics['pass_rate'] * 100,
            'reliability': (1 - metrics['error_rate']) * 100,
            'flakiness': (1 - metrics['flaky_test_pct']) * 100,
        }

        return {
            'overall': sum(components.values()) / len(components),
            'components': components
        }

Risk Register

High Risk: Data Quality & Garbage-In-Garbage-Out

Description: If normalized data is incorrect, all ML models and insights will be unreliable.

Mitigation: Implement data quality checks (dbt tests, Great Expectations), manual validation of first 1000 events per source, side-by-side comparison with original sources before going live.

Owner: Data Engineer

High Risk: Model Accuracy & False Positives

Description: Anomaly detector may flag legitimate changes as issues, overwhelming teams.

Mitigation: Conservative initial thresholds, manual review mode for first month, calibration with business team, integration of calendar/holiday data for known variations.

Owner: ML Engineer

Medium Risk: Performance at Scale

Description: Dashboard queries may slow down with 6+ months of historical data.

Mitigation: TimescaleDB compression for old data, materialized views for aggregates, query profiling and optimization, pagination for large result sets.

Owner: Backend Lead

Medium Risk: Integration Complexity

Description: Jira/GitHub integrations may require custom authentication and error handling.

Mitigation: Use official API libraries, comprehensive error handling, rate limiting, mock integration tests, gradual rollout with pilot team first.

Owner: Backend Lead
