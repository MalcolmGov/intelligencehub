import asyncio

async def analyze_broken_selector(
    selector: str,
    element_html: str,
    page_context: str
) -> dict:
    """Analyze broken selector and propose repairs."""

    prompt = f"""Analyze this broken CSS/XPath selector and suggest repairs:

Broken Selector: {selector}
Element HTML: {element_html}
Page Context: {page_context}

Return a JSON object with:
- repair_candidates: List of 3 alternative selectors
- confidence_scores: Confidence for each candidate (0-100)
- repair_method: "semantic" | "cv" | "aria" | "combination"
- explanation: Why the selector broke and how repairs work
- approval_required: true"""

    msg = await asyncio.to_thread(
        client.messages.create,
        model="gpt-4",
        max_tokens=500,
        messages=[{"role": "user", "content": prompt}]
    )

    import json
    response_text = msg.content[0].text
    return json.loads(response_text)

export default function HealingDashboard() {
  const [proposals, setProposals] = useState([])
  const [filter, setFilter] = useState('pending')
  const [selectedProposal, setSelectedProposal] = useState(null)

  useEffect(() => {
    fetchProposals()
  }, [filter])

  const fetchProposals = async () => {
    const response = await fetch(`/api/healing/proposals?status=${filter}`)
    const data = await response.json()
    setProposals(data.proposals)
  }

  const approveProposal = async (id: string, selectedIndex: number) => {
    const proposal = proposals.find(p => p.id === id)
    if (!proposal) return

    const response = await fetch(`/api/healing/approve`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        proposal_id: id,
        selected_selector: proposal.proposed_selectors[selectedIndex],
        test_id: proposal.test_id
      })
    })

    if (response.ok) {
      setProposals(proposals.map(p =>
        p.id === id ? { ...p, status: 'approved' } : p
      ))
      setSelectedProposal(null)
    }
  }

  const rejectProposal = async (id: string) => {
    await fetch(`/api/healing/reject`, {
      method: 'POST',
